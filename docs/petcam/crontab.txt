# There are no petcam-detect lines any more: the detector runs on Ralph as
# petcam-detect.service. The app (petcam.service) is Ralph's too; Tom serves

PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin:/usr/local/go/bin
# Health sample into the journal: throttled flags latch until a power cycle, so
# this is the only way to catch a brownout that reboots the Pi.
*/2 * * * * logger -t tom-health "up=$(cut -d. -f1 /proc/uptime)s $(vcgencmd get_throttled) temp=$(vcgencmd measure_temp) ext5v=$(vcgencmd pmic_read_adc | awk '/EXT5V_V/{print $2}')"
# Watchdogs (every 2 min): the capture supervisor relaunches a dead pipeline.
*/2 * * * * pgrep -f "petca[m].sh$" >/dev/null 2>&1 || nohup /home/tom/projects/homelab/scripts/petcam.sh >/tmp/petcam.log 2>&1 &
@reboot /home/tom/projects/homelab/scripts/petcam.sh >/tmp/petcam.log 2>&1 &
*/2 * * * * sudo /usr/local/bin/wifi-guard.sh >/dev/null 2>&1
